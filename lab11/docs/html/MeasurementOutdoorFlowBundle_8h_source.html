68template <typename Scalar>
69Eigen::Matrix<Scalar, 3, Eigen::Dynamic>
70MeasurementOutdoorFlowBundle::predictFlowImpl(const Eigen::VectorX<Scalar> & x,
71 const Eigen::Matrix<double, 3, Eigen::Dynamic> & pkm1,
72 const Eigen::Matrix<double, 3, Eigen::Dynamic> & pk) const
73{
74 (void)pk; // not used for the prediction branch decision (only pkm1 ray & pose are needed)
75
76 assert(x.rows() >= 18);
77 assert(x.cols() == 1);
78 assert(pkm1.cols() == pk.cols());
79
80 using Mat3 = Eigen::Matrix<Scalar,3,3>;
81 using Vec3 = Eigen::Matrix<Scalar,3,1>;
82
83 // State layout: [nu(6)=0..5, eta_k(6)=6..11, eta_km1(6)=12..17]
84 const Eigen::Matrix<Scalar,6,1> etak = x.template segment<6>(6);
85 const Eigen::Matrix<Scalar,6,1> etakm1 = x.template segment<6>(12);
86
87 // Poses and fixed Tbc
88 const Pose<Scalar> Tnb_k ( rpy2rot(etak .template tail<3>()), etak .template head<3>() );
89 const Pose<Scalar> Tnb_km1 ( rpy2rot(etakm1.template tail<3>()), etakm1.template head<3>() );
90 const Pose<Scalar> Tbc = camera_.Tbc;
91
92 const Pose<Scalar> Tnc_k = Tnb_k * Tbc; // world->cam at k
93 const Pose<Scalar> Tnc_km1 = Tnb_km1 * Tbc; // world->cam at k-1
94
95 const Eigen::Matrix<Scalar,3,3> R_WCk = Tnc_k .rotationMatrix;
96 const Eigen::Matrix<Scalar,3,1> r_WCk = Tnc_k .translationVector;
97 const Eigen::Matrix<Scalar,3,3> R_WCkm1 = Tnc_km1.rotationMatrix;
98 const Eigen::Matrix<Scalar,3,1> r_WCkm1 = Tnc_km1.translationVector;
99
100 const Eigen::Matrix<Scalar,3,3> R_CkCkm1 = R_WCk.transpose() * R_WCkm1;
101
102 // Build K^{-1} explicitly (avoids autodiff .inverse)
103 const double fx = camera_.cameraMatrix.at<double>(0,0);
104 const double fy = camera_.cameraMatrix.at<double>(1,1);
105 const double cx = camera_.cameraMatrix.at<double>(0,2);
106 const double cy = camera_.cameraMatrix.at<double>(1,2);
107
108 Mat3 Kinv;
109 Kinv << Scalar(1.0 / fx), Scalar(0), Scalar(-cx / fx),
110 Scalar(0), Scalar(1.0 / fy), Scalar(-cy / fy),
111 Scalar(0), Scalar(0), Scalar(1);
112
113 Eigen::Matrix<Scalar,3,Eigen::Dynamic> pk_hat(3, pkm1.cols());
114
115 const double eps2 = 1e-12; // guard for near-horizon
116
117 for (int j = 0; j < pkm1.cols(); ++j)
118 {
119 // pkm1 holds UNDISTORTED pixels [ū, v̄, 1]; turn into a ray via K^{-1}
120 const Vec3 pUbar_km1( Scalar(pkm1(0,j)), Scalar(pkm1(1,j)), Scalar(1) );
121 const Vec3 dC_km1 = Kinv * pUbar_km1; // direction in C_{k-1}
122 const Vec3 dW_km1 = R_WCkm1 * dC_km1; // direction in world
123
124 // Branch in double to keep autodiff stable, then compute Scalar path
125 const Scalar denom = dW_km1.z();
126 const double denom_val = static_cast<double>(denom);
127 const double denom2_val = denom_val * denom_val;
128 const bool nondeg = (denom2_val > eps2);
129
130 if (nondeg) {
131 const Scalar s = -r_WCkm1.z() / denom;
132 const double s_val = static_cast<double>(s);
133 const bool front = (s_val > 0.0);
134 if (front) {
135 const Vec3 PW = r_WCkm1 + s * dW_km1;
136 const Vec3 PC_k = R_WCk.transpose() * (PW - r_WCk);
137 pk_hat.col(j) = PC_k; // homogeneous in C_k (z is depth)
138 continue;
139 }
140 }
141
142 // Sky/degenerate → plane at infinity: rotate direction only
143 pk_hat.col(j) = R_CkCkm1 * dC_km1;
144 }
145 return pk_hat;
146}

148template <typename Scalar>
149Scalar MeasurementOutdoorFlowBundle::logLikelihoodImpl(const Eigen::VectorX<Scalar> & x) const
150{
151 assert(pkm1_.cols() == pk_.cols());
152
153 // Predict *undistorted* rays in C_k
154 Eigen::Matrix<Scalar,3,Eigen::Dynamic> pk_hat = predictFlowImpl(x, pkm1_, pk_);
155
156 // Dehomogenize to normalized image coords
157 Eigen::Matrix<Scalar,2,Eigen::Dynamic> rNorm_hat(2, pk_hat.cols());
158 const Scalar eps = Scalar(1e-9);
159 for (int j = 0; j < pk_hat.cols(); ++j) {
160 Scalar z = pk_hat(2,j);
161 if (Eigen::numext::abs(z) < eps) z = (z >= Scalar(0)) ? eps : -eps;
162 rNorm_hat(0,j) = pk_hat(0,j) / z;
163 rNorm_hat(1,j) = pk_hat(1,j) / z;
164 }
165
166 // Map normalized -> UNDISTORTED *pixels*
167 const double fx = camera_.cameraMatrix.at<double>(0,0);
168 const double fy = camera_.cameraMatrix.at<double>(1,1);
169 const double cx = camera_.cameraMatrix.at<double>(0,2);
170 const double cy = camera_.cameraMatrix.at<double>(1,2);
171
172 Eigen::Matrix<Scalar,2,Eigen::Dynamic> rUbar_hat(2, rNorm_hat.cols());
173 rUbar_hat.row(0) = rNorm_hat.row(0).array() * Scalar(fx) + Scalar(cx);
174 rUbar_hat.row(1) = rNorm_hat.row(1).array() * Scalar(fy) + Scalar(cy);
175
176 // Measured UNDISTORTED pixels are pk_.topRows<2>()
177 Eigen::Matrix<Scalar,2,Eigen::Dynamic> rUbar_meas = pk_.template topRows<2>().template cast<Scalar>();
178
179 const Scalar inv_s2 = Scalar(1) / Scalar(sigma_ * sigma_); // σ in pixels
180 Scalar logLik = Scalar(0);
181 for (int j = 0; j < rUbar_hat.cols(); ++j) {
182 const Scalar dx = rUbar_hat(0,j) - rUbar_meas(0,j);
183 const Scalar dy = rUbar_hat(1,j) - rUbar_meas(1,j);
184 logLik -= Scalar(0.5) * (dx*dx + dy*dy) * inv_s2;
185 }
186 return logLik;
187}
